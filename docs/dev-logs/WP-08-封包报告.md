## 4. 审计复算指引

```bash
# 本 WP 测试(任意平台;Kali 上 dpkg 断言已平台自适应)
.venv/bin/python -m pytest tests/test_toolmap.py -v

# lint
.venv/bin/ruff check src/foam/agent/toolmap.py src/foam/agent/tool_catalog.py tests/test_toolmap.py

# 1884B 复现(PYTHONPATH=src;缺失集 = §3.2 的 41 个名字)
PYTHONPATH=src python3 - <<'EOF'
from foam.agent.tool_catalog import TOOL_CATALOG
from foam.agent.toolmap import ScanResult, ToolStatus, render_tool_map, render_startup_line
missing = {
    "rustscan","naabu","sublist3r","assetfinder","mtr","enum4linux-ng",
    "joomscan","droopescan","lynis","tplmap","wig",
    "feroxbuster","katana","hakrawler","gau","waybackurls","xsstrike","arjun","jwt_tool",
    "crowbar","name-that-hash","cupp","fcrackzip","pdfcrack",
    "bettercap","mdk4","ghidra","gdb","foremost","steghide","volatility3",
    "ncat","rlwrap","chisel","sshuttle","crackmapexec","bloodhound",
    "eyewitness","gowitness","dradis","keepassxc",
}
statuses = tuple(
    ToolStatus(entry=e, group=g.key, available=e.name not in missing,
               via=None if e.name in missing else "which",
               location=None if e.name in missing else f"/usr/bin/{e.name}")
    for g in TOOL_CATALOG for e in g.tools
)
scan = ScanResult(catalog=TOOL_CATALOG, statuses=statuses, dpkg_available=True)
print(render_startup_line(scan))
print(len(render_tool_map(scan).encode()))  # 期望 1884
EOF
```

审计要点建议:① 目录 137 条的 name 是否即探测键(无第二字段可漂移);
② 降级层级 3→0 的字节单调性(有单测);③ 注入契约是否经真实
`build_system_prompt` 而非拼字符串模拟;④ 缺失标注的 apt 包名出处
(`ToolEntry.package`);⑤ Kali 数字与 §3 原始输出是否互洽。
